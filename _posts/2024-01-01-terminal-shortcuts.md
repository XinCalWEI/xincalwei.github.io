  <!-- Title -->
  <h1>🎴 Common Terminal Shortcuts</h1>
  <p class="text-muted">A clean, card-style cheatsheet for frequently used CLI shortcuts.</p>

  <!-- Navigation -->
  <div class="card my-3">
    <div class="card-body">
      <h3 class="card-title">🏁 Navigation</h3>

      <p class="card-text">
        ⌨️ <strong>Control + A</strong><br>
        Jump to the beginning of the line.
      </p>

      <p class="card-text">
        ⌨️ <strong>Control + E</strong><br>
        Jump to the end of the line.
      </p>

      <p class="card-text">
        ⌨️ <strong>Option + ←</strong><br>
        Move cursor left by one word.
      </p>

      <p class="card-text">
        ⌨️ <strong>Option + →</strong><br>
        Move cursor right by one word.
      </p>
    </div>

  </div>

  <!-- Deleting Text -->
  <div class="card my-3">
    <div class="card-body">
      <h3 class="card-title">✂️ Deleting Text</h3>

      <p class="card-text">
        ⌨️ <strong>Control + U</strong><br>
        Delete everything before the cursor.
      </p>

      <p class="card-text">
        ⌨️ <strong>Control + K</strong><br>
        Delete everything after the cursor (including character under cursor).
      </p>

      <p class="card-text">
        ⌨️ <strong>Control + A + K</strong><br>
        Delete the entire line.
      </p>

      <p class="card-text">
        ⌨️ <strong>Control + W</strong><br>
        Delete the word before the cursor.
      </p>

      <p class="card-text">
        ⌨️ <strong>Esc + D</strong><br>
        Delete the word after the cursor.
      </p>

      <p class="card-text">
        ⌨️ <strong>Control + D</strong><br>
        Delete character under cursor; exits if nothing to delete.
      </p>
    </div>

  </div>

  <!-- Clearing / Stopping -->
  <div class="card my-3">
    <div class="card-body">
      <h3 class="card-title">🧹 Clearing & Stopping</h3>

      <p class="card-text">
        ⌨️ <strong>Control + L</strong><br>
        Clear the screen.
      </p>

      <p class="card-text">
        🔄 <strong>clear</strong><br>
        Clear the screen.
      </p>

      <p class="card-text">
        ⌨️ <strong>Command + K</strong> (macOS)<br>
        Clear the screen.
      </p>

      <p class="card-text">
        ⌨️ <strong>Control + C</strong><br>
        Terminate / interrupt the current command.
      </p>
    </div>

  </div>

  <!-- Yank / Paste -->
  <div class="card my-3">
    <div class="card-body">
      <h3 class="card-title">📋 Paste / Yank</h3>

      <p class="card-text">
        ⌨️ <strong>Control + Y</strong><br>
        Paste ("yank") the last deleted text. Works with <code>Ctrl + U</code> and <code>Ctrl + K</code>.
      </p>
    </div>

  </div>

</div>
